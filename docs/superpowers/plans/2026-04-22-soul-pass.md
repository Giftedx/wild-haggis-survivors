```typescript
let recordingDestination: MediaStreamAudioDestinationNode | null = null;

/**
 * W27 Phase 3 — expose an audio MediaStream that captures everything
 * routed through the shared output compressor. Parallel tap; does not
 * interfere with playback to `ctx.destination`.
 *
 * Returns null if AudioContext isn't available (pre-user-gesture or
 * unsupported browser). Caller tears down via `disposeRecordingAudioStream`.
 */
export function createRecordingAudioStream(): MediaStream | null {
  const ctx = getAudioContext();
  if (!ctx) return null;
  if (recordingDestination) return recordingDestination.stream;
  try {
    recordingDestination = ctx.createMediaStreamDestination();
    const output = getOutputNode();
    output?.connect(recordingDestination);
    return recordingDestination.stream;
  } catch {
    recordingDestination = null;
    return null;
  }
}

export function disposeRecordingAudioStream(): void {
  if (!recordingDestination) return;
  try {
    const output = getOutputNode();
    output?.disconnect(recordingDestination);
  } catch { /* already disconnected */ }
  recordingDestination = null;
}
```

Adapt the `getAudioContext()` / `getOutputNode()` names to match whatever the module already exports.

Change the `start()` signature and body:

```typescript
  start(audioStream?: MediaStream): void {
    if (this.running || !this.isAvailable() || !this.mimeType) return;
    try {
      const videoStream = (this.canvas as unknown as {
        captureStream: (fps?: number) => MediaStream;
      }).captureStream(this.fps);
      this.stream = videoStream;

      const streamWithAudio = audioStream
        ? new MediaStream([
            ...videoStream.getVideoTracks(),
            ...audioStream.getAudioTracks(),
          ])
        : videoStream;

      try {
        const recorder = new MediaRecorder(streamWithAudio, { mimeType: this.mimeType });
        this.recorder = recorder;
        this.audioAttached = audioStream != null && audioStream.getAudioTracks().length > 0;
        this.wireRecorder(recorder);
        recorder.start(this.timesliceMs);
        this.running = true;
        return;
      } catch {
        // Safari multi-track fallback — retry with canvas-only stream.
        if (audioStream) {
          const recorder = new MediaRecorder(videoStream, { mimeType: this.mimeType });
          this.recorder = recorder;
          this.audioAttached = false;
          this.wireRecorder(recorder);
          recorder.start(this.timesliceMs);
          this.running = true;
          return;
        }
        throw new Error('MediaRecorder construct failed');
      }
    } catch {
      this.running = false;
    }
  }

  hasAudio(): boolean {
    return this.audioAttached && this.running;
  }

  private wireRecorder(recorder: MediaRecorder): void {
    recorder.ondataavailable = (e: BlobEvent) => {
      if (!e.data || e.data.size === 0) return;
      this.buffer.push(e.data);
      if (this.buffer.length > this.capacity) {
        this.buffer.splice(0, this.buffer.length - this.capacity);
      }
    };
  }
```

Update `stop()` to reset `audioAttached`:

```typescript
  stop(): void {
    if (!this.running) return;
    try {
      this.recorder?.stop();
    } catch { /* already stopped */ }
    this.recorder = null;
    this.stream?.getTracks().forEach((t) => t.stop());
    this.stream = null;
    this.running = false;
    this.audioAttached = false;
    this.buffer = [];
  }
```

```typescript
// src/art/sprites/wildlife/hare.ts
import Phaser from 'phaser';

const HARE_BODY = 0x8a6f4a;      // warm brown
const HARE_BODY_DARK = 0x5f4d33;
const HARE_BODY_LIGHT = 0xb09172;
const HARE_TAIL = 0xf2ebd5;      // cream
const HARE_EYE = 0x1a1410;
const HARE_OUTLINE = 0x2a1e12;

/**
 * W91 ambient hare — small terrestrial creature.
 * Frame 0 = idle (crouched), Frame 1 = mid-hop (arched).
 */
export function drawHare(
  g: Phaser.GameObjects.Graphics,
  cx: number,
  cy: number,
  frame: number = 0,
): void {
  const hopping = frame === 1;
  const bodyY = hopping ? cy - 4 : cy;

  // Outline
  g.fillStyle(HARE_OUTLINE, 1);
  g.fillEllipse(cx, bodyY + 1, 16, 10);

  // Body
  g.fillStyle(HARE_BODY, 1);
  g.fillEllipse(cx, bodyY, 14, 8);

  // Belly highlight
  g.fillStyle(HARE_BODY_LIGHT, 1);
  g.fillEllipse(cx, bodyY + 2, 10, 5);

  // Head (right-facing)
  g.fillStyle(HARE_BODY, 1);
  g.fillEllipse(cx + 6, bodyY - 2, 6, 5);

  // Ears (two upright lines)
  g.fillStyle(HARE_BODY_DARK, 1);
  g.fillRect(cx + 5, bodyY - 7, 1, 4);
  g.fillRect(cx + 7, bodyY - 7, 1, 4);

  // Eye
  g.fillStyle(HARE_EYE, 1);
  g.fillRect(cx + 7, bodyY - 3, 1, 1);

  // Tail (cream flash — prominent when hopping)
  g.fillStyle(HARE_TAIL, 1);
  g.fillCircle(cx - 6, bodyY, hopping ? 2 : 1.5);

  // Legs (tucked at idle, extended at hop)
  g.fillStyle(HARE_BODY_DARK, 1);
  if (hopping) {
    g.fillRect(cx - 2, bodyY + 3, 2, 5);
    g.fillRect(cx + 3, bodyY + 3, 2, 5);
  } else {
    g.fillRect(cx - 2, bodyY + 4, 2, 3);
    g.fillRect(cx + 3, bodyY + 4, 2, 3);
  }
}
```

```typescript
// src/entities/wildlife/ambientCreatureState.ts
export type AmbientCreatureMode = 'wander' | 'flee';

export interface AmbientCreatureState {
  mode: AmbientCreatureMode;
  fleeEndsAtMs: number;
}

export interface TickInput {
  nowMs: number;
  creatureX: number;
  creatureY: number;
  playerX: number;
  playerY: number;
  nearestEnemyDistSq: number;
  fleeRadius: number;
  enemyFleeRadius: number;
  fleeDurationMs: number;
}

export function initAmbientCreatureState(nowMs: number): AmbientCreatureState {
  return { mode: 'wander', fleeEndsAtMs: nowMs };
}

function threatPresent(input: TickInput): boolean {
  const dx = input.playerX - input.creatureX;
  const dy = input.playerY - input.creatureY;
  const playerDistSq = dx * dx + dy * dy;
  if (playerDistSq < input.fleeRadius * input.fleeRadius) return true;
  if (
    input.enemyFleeRadius > 0 &&
    input.nearestEnemyDistSq < input.enemyFleeRadius * input.enemyFleeRadius
  ) {
    return true;
  }
  return false;
}

export function tickAmbientCreatureState(
  state: AmbientCreatureState,
  input: TickInput,
): AmbientCreatureState {
  const threat = threatPresent(input);

  if (state.mode === 'wander') {
    if (threat) {
      return { mode: 'flee', fleeEndsAtMs: input.nowMs + input.fleeDurationMs };
    }
    return state;
  }

  // mode === 'flee'
  if (threat) {
    // Threat present — keep fleeing. If timer already expired this tick,
    // extend so the creature doesn't pop back to wander with threat still close.
    if (input.nowMs >= state.fleeEndsAtMs) {
      return { mode: 'flee', fleeEndsAtMs: input.nowMs + input.fleeDurationMs };
    }
    return state;
  }

  // No threat. If timer expired, go back to wander.
  if (input.nowMs >= state.fleeEndsAtMs) {
    return { mode: 'wander', fleeEndsAtMs: state.fleeEndsAtMs };
  }
  return state;
}
```

```typescript
// src/systems/wildlifeSpawnRoll.ts
import type { BiomeId } from '@/data/biomes';
import { WILDLIFE_DEFS, WILDLIFE_KEYS, type WildlifeKey } from '@/data/wildlife';

export interface SpawnRollInput {
  biome: BiomeId;
  skipChance: number;    // 0..1 probability of skipping this roll
  rngSample: number;     // 0..1 — caller-provided RNG sample (for test determinism)
}

export interface SpawnRollResult {
  key: WildlifeKey;
}

export function rollWildlifeSpawn(input: SpawnRollInput): SpawnRollResult | null {
  if (input.rngSample < input.skipChance) return null;

  // Build a weighted list of non-zero-weight creatures for this biome.
  const entries: Array<{ key: WildlifeKey; weight: number }> = [];
  let total = 0;
  for (const key of WILDLIFE_KEYS) {
    const w = WILDLIFE_DEFS[key].biomeWeights[input.biome] ?? 0;
    if (w > 0) {
      entries.push({ key, weight: w });
      total += w;
    }
  }
  if (total === 0) return null;

  // Re-sample in [0, 1] using the fraction above skipChance.
  const rescaled = (input.rngSample - input.skipChance) / (1 - input.skipChance);
  let acc = 0;
  const target = rescaled * total;
  for (const entry of entries) {
    acc += entry.weight;
    if (target < acc) return { key: entry.key };
  }
  return { key: entries[entries.length - 1]!.key };
}
```

```typescript
import type { WildlifeDef } from '@/data/wildlife';
import {
  type AmbientCreatureState,
  initAmbientCreatureState,
  tickAmbientCreatureState,
} from './ambientCreatureState';

export class AmbientCreature {
  public sprite: Phaser.GameObjects.Sprite;
  public def: WildlifeDef;
  private state: AmbientCreatureState;
  private wanderAngle: number;
  private wanderAngleChangeAtMs: number;

  constructor(scene: Phaser.Scene, def: WildlifeDef, x: number, y: number, nowMs: number) {
    this.def = def;
    this.sprite = scene.add.sprite(x, y, def.spriteKeyIdle);
    this.sprite.setScale(def.scale);
    this.sprite.setDepth(def.aerial ? 450 : 50); // 50 = below entities; 450 = above gameplay layer
    this.state = initAmbientCreatureState(nowMs);
    this.wanderAngle = Math.random() * Math.PI * 2;
    this.wanderAngleChangeAtMs = nowMs + 1000 + Math.random() * 3000;
  }

  update(nowMs: number, dtSec: number, playerX: number, playerY: number, nearestEnemyDistSq: number): void {
    // Tick state machine
    const next = tickAmbientCreatureState(this.state, {
      nowMs,
      creatureX: this.sprite.x,
      creatureY: this.sprite.y,
      playerX,
      playerY,
      nearestEnemyDistSq,
      fleeRadius: this.def.fleeRadius,
      enemyFleeRadius: this.def.enemyFleeRadius,
      fleeDurationMs: 2000,
    });

    // Update movement based on mode
    let dx = 0, dy = 0;
    if (next.mode === 'flee') {
      dx = this.sprite.x - playerX;
      dy = this.sprite.y - playerY;
      const len = Math.hypot(dx, dy) || 1;
      dx = (dx / len) * this.def.fleeSpeed * dtSec;
      dy = (dy / len) * this.def.fleeSpeed * dtSec;
      this.sprite.setTexture(this.def.spriteKeyMove);
    } else {
      if (nowMs >= this.wanderAngleChangeAtMs) {
        this.wanderAngle = Math.random() * Math.PI * 2;
        this.wanderAngleChangeAtMs = nowMs + 1500 + Math.random() * 3500;
      }
      dx = Math.cos(this.wanderAngle) * this.def.baseSpeed * dtSec;
      dy = Math.sin(this.wanderAngle) * this.def.baseSpeed * dtSec;
      // Alternate idle/move every 500ms for visual rhythm
      const moving = Math.floor(nowMs / 500) % 2 === 0;
      this.sprite.setTexture(moving ? this.def.spriteKeyMove : this.def.spriteKeyIdle);
    }

    this.sprite.x += dx;
    this.sprite.y += dy;

    // Flip sprite toward movement direction (terrestrial only — buzzard keeps its symmetry)
    if (!this.def.aerial && dx !== 0) {
      this.sprite.setFlipX(dx < 0);
    }

    this.state = next;
  }

  isOffscreen(cameraX: number, cameraY: number, cameraW: number, cameraH: number, margin: number): boolean {
    return (
      this.sprite.x < cameraX - margin ||
      this.sprite.x > cameraX + cameraW + margin ||
      this.sprite.y < cameraY - margin ||
      this.sprite.y > cameraY + cameraH + margin
    );
  }

  destroy(): void {
    this.sprite.destroy();
  }
}
```

```typescript
// src/systems/WildlifeSystem.ts
import Phaser from 'phaser';
import { AmbientCreature } from '@/entities/wildlife/AmbientCreature';
import { WILDLIFE_DEFS } from '@/data/wildlife';
import { rollWildlifeSpawn } from './wildlifeSpawnRoll';
import type { BiomeManager } from './biome/BiomeManager';

const MAX_ACTIVE = 5;
const OFFSCREEN_DESPAWN_MARGIN = 500;
const SPAWN_INTERVAL_MIN_MS = 3000;
const SPAWN_INTERVAL_MAX_MS = 5000;
const SKIP_CHANCE = 0.15;

export class WildlifeSystem {
  private scene: Phaser.Scene;
  private biomeManager: BiomeManager;
  private reduceParticles: () => boolean;
  private creatures: AmbientCreature[] = [];
  private nextSpawnAtMs: number;

  constructor(
    scene: Phaser.Scene,
    biomeManager: BiomeManager,
    reduceParticles: () => boolean,
  ) {
    this.scene = scene;
    this.biomeManager = biomeManager;
    this.reduceParticles = reduceParticles;
    this.nextSpawnAtMs = 0;
  }

  update(nowMs: number, dtSec: number, playerX: number, playerY: number, enemies: Phaser.GameObjects.Group): void {
    // Opt-out: clear everything and bail.
    if (this.reduceParticles()) {
      this.clearAll();
      return;
    }

    // Spawn cadence
    if (nowMs >= this.nextSpawnAtMs && this.creatures.length < MAX_ACTIVE) {
      this.trySpawn(nowMs, playerX, playerY);
      this.nextSpawnAtMs = nowMs + SPAWN_INTERVAL_MIN_MS +
        Math.random() * (SPAWN_INTERVAL_MAX_MS - SPAWN_INTERVAL_MIN_MS);
    }

    // Update + cull off-screen
    const cam = this.scene.cameras.main;
    const enemyArray = enemies.getChildren() as Array<Phaser.GameObjects.GameObject & { x: number; y: number; active: boolean }>;

    for (let i = this.creatures.length - 1; i >= 0; i--) {
      const c = this.creatures[i]!;

      // Nearest enemy squared distance (skip for aerial)
      let nearestSq = Infinity;
      if (!c.def.aerial) {
        for (const e of enemyArray) {
          if (!e.active) continue;
          const ex = e.x - c.sprite.x;
          const ey = e.y - c.sprite.y;
          const d2 = ex * ex + ey * ey;
          if (d2 < nearestSq) nearestSq = d2;
        }
      }

      c.update(nowMs, dtSec, playerX, playerY, nearestSq);

      if (c.isOffscreen(cam.worldView.x, cam.worldView.y, cam.worldView.width, cam.worldView.height, OFFSCREEN_DESPAWN_MARGIN)) {
        c.destroy();
        this.creatures.splice(i, 1);
      }
    }
  }

  shutdown(): void {
    this.clearAll();
  }

  getActiveCount(): number {
    return this.creatures.length;
  }

  private trySpawn(nowMs: number, playerX: number, playerY: number): void {
    const biome = this.biomeManager.biomeAt(playerX, playerY);
    const roll = rollWildlifeSpawn({
      biome,
      skipChance: SKIP_CHANCE,
      rngSample: Math.random(),
    });
    if (!roll) return;

    const def = WILDLIFE_DEFS[roll.key];
    // Spawn off-screen in a random direction relative to camera
    const cam = this.scene.cameras.main;
    const cx = cam.worldView.x + cam.worldView.width / 2;
    const cy = cam.worldView.y + cam.worldView.height / 2;
    const angle = Math.random() * Math.PI * 2;
    const radius = cam.worldView.width * 0.6 + Math.random() * 100;
    const x = cx + Math.cos(angle) * radius;
    const y = cy + Math.sin(angle) * radius;

    this.creatures.push(new AmbientCreature(this.scene, def, x, y, nowMs));
  }

  private clearAll(): void {
    for (const c of this.creatures) c.destroy();
    this.creatures = [];
  }
}
```
